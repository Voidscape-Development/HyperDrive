query EventStandingsQuery($eventSlug: String!, $perPage: Int!) {
	event(slug: $eventSlug) {
		state
		phases {
			bracketType
		}
		standings(query: {page: 1, perPage: $perPage}) {
			nodes {
				placement
				isFinal
				entrant {
					id
					name
					initialSeedNum
					participants {
						player {
							id
							gamerTag
							prefix
						}
					}
				}
			}
		}
	}
}
