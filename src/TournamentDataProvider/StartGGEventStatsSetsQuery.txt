query EventStatsSetsQuery($eventSlug: String!, $page: Int!, $perPage: Int!) {
	event(slug: $eventSlug) {
		sets(page: $page, perPage: $perPage, sortType: RECENT, filters: {state: 3}) {
			pageInfo {
				total
				totalPages
			}
			nodes {
				id
				startedAt
				completedAt
				totalGames
				round
				fullRoundText
				winnerId
				entrant1Score
				entrant2Score
				slots {
					entrant {
						id
						name
						initialSeedNum
						participants {
							player {
								id
								gamerTag
								prefix
							}
						}
					}
				}
				games {
					winnerId
					orderNum
					stage {
						id
						name
					}
					selections {
						entrant {
							id
						}
						selectionValue
					}
				}
				phaseGroup {
					displayIdentifier
					phase {
						id
						name
						bracketType
					}
				}
			}
		}
	}
}
