query LeagueQuery($eventSlug: String, $leagueSlug: String, $byEvent: Boolean!, $perPage: Int!) {
	event(slug: $eventSlug) @include(if: $byEvent) {
		league {
			...league
		}
	}
	league(slug: $leagueSlug) @skip(if: $byEvent) {
		...league
	}
}

fragment league on League {
	id
	name
	slug
	images {
		type
		url
	}
	standings(query: {page: 1, perPage: $perPage}) {
		pageInfo {
			total
		}
		nodes {
			placement
			totalPoints
			player {
				id
				gamerTag
				prefix
			}
			entrant {
				name
			}
		}
	}
}
