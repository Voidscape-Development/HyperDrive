query SetQuery($id: ID!, $profile: Boolean = true) {
	set(id: $id) {
		id
		event {
			hasTasks
			videogame {
				id
			}
			isOnline
		}
		fullRoundText
		round
		state
		totalGames
		entrant1Score
		entrant2Score
		slots {
			entrant {
				id
				name
				initialSeedNum
				standing {
					setRecordWithoutByes
				}
				participants {
					id
					user {
						id
						slug
						name
						genderPronoun @include(if: $profile)
						authorizations @include(if: $profile) {
							type
							externalUsername
						}
						location @include(if: $profile) {
							city
							country
							state
						}
						images(type: "profile") {
							url
						}
					}
					player {
						id
						gamerTag
						prefix
					}
				}
			}
		}
		games {
			orderNum
			stage {
				id
			}
			selections {
				entrant {
					id
					participants {
						player {
							id
						}
					}
				}
				selectionValue
			}
			winnerId
		}
		round
		phaseGroup {
			phase {
				name
				groupCount
				bracketType
                numSeeds
                seedNum
				progressionsOut {
					id
				}
			}
			displayIdentifier
		}
		stream {
			streamName
			streamSource
		}
	}
}
