query TournamentHistoryDataQuery($playerID: ID!, $gameID: ID!) {
	player(id: $playerID) {
		recentStandings(videogameId: $gameID, limit: 10) {
			placement
			entrant {
				initialSeedNum
				event {
					name
					numEntrants
					startAt
					phases {
						bracketType
					}
					tournament {
						name
						venueAddress
						startAt
						images(type: "profile") {
							url
						}
					}
				}
			}
		}
	}
}
